  <script>
    let currentScope = 'unit';
    let countLolos = 1;
    let countReject = 1;

    function switchTab(tab) {
      document.querySelectorAll('.nav-button').forEach(btn => btn.classList.remove('active'));
      document.getElementById('tabViewMaterial').style.display = 'none';
      document.getElementById('tabViewCeklist').style.display = 'none';
      document.getElementById('tabViewSuratJalan').style.display = 'none';

      if (tab === 'material') {
        document.getElementById('navMat').classList.add('active');
        document.getElementById('tabViewMaterial').style.display = 'block';
      } else if (tab === 'ceklist') {
        document.getElementById('navChk').classList.add('active');
        document.getElementById('tabViewCeklist').style.display = 'block';
      } else if (tab === 'suratjalan') {
        document.getElementById('navSJ').classList.add('active');
        document.getElementById('tabViewSuratJalan').style.display = 'block';
      }
    }

    // Modal Input Kedatangan Material
    function openInputModal() {
      document.getElementById('modalInput').style.display = 'flex';
    }
    function closeInputModal() {
      document.getElementById('modalInput').style.display = 'none';
    }
    function saveNewDelivery() {
      const mat = document.getElementById('modalMat').value;
      const qty = document.getElementById('modalQty').value;
      const sj = document.getElementById('modalSJ').value;

      if (!qty || !sj) {
        alert('Mohon lengkapi Jumlah Masuk dan Nomor Surat Jalan!');
        return;
      }
      alert(`Pencatatan Berhasil!\n\nMaterial: ${mat}\nQty: ${qty}\nNo. SJ: ${sj}`);
      closeInputModal();
    }

    // Modal Ceklist Lapangan
    function openChecklistModal() {
      document.getElementById('modalChecklist').style.display = 'flex';
    }
    function closeChecklistModal() {
      document.getElementById('modalChecklist').style.display = 'none';
    }

    function setScope(scope) {
      currentScope = scope;
      const btnUnit = document.getElementById('btnScopeUnit');
      const btnShaft = document.getElementById('btnScopeShaft');
      const areaUnit = document.getElementById('areaUnit');
      const areaShaft = document.getElementById('areaShaft');

      if (scope === 'unit') {
        btnUnit.classList.add('active');
        btnShaft.classList.remove('active');
        areaUnit.style.display = 'block';
        areaShaft.style.display = 'none';
      } else {
        btnShaft.classList.add('active');
        btnUnit.classList.remove('active');
        areaUnit.style.display = 'none';
        areaShaft.style.display = 'block';
      }
    }

    function saveNewChecklist() {
      const lantai = document.getElementById('formLantai').value;
      const status = document.getElementById('formStatus').value;
      const progres = document.getElementById('formProgres').value;
      const spv = document.getElementById('formSPV').value || 'Inspektor';
      
      let badgeText = "";
      let itemTitle = "";

      if (currentScope === 'unit') {
        const tipe = document.getElementById('formTipe').value;
        const kamar = document.getElementById('formKamar').value || '-';
        const kategori = document.getElementById('formKategoriMEP').value;
        badgeText = `${lantai.toUpperCase()} - ${tipe}`;
        itemTitle = `${kategori} (Kamar ${kamar})`;
      } else {
        const shaft = document.getElementById('formShaftNo').value;
        const itemShaft = document.getElementById('formShaftItem').value;
        badgeText = `${lantai.toUpperCase()} - ${shaft.toUpperCase()}`;
        itemTitle = itemShaft;
      }

      // Update Metrik
      if (status === 'OK') {
        countLolos++;
        document.getElementById('metricLolos').innerText = countLolos;
      } else {
        countReject++;
        document.getElementById('metricReject').innerText = countReject;
      }

      const isOk = status === 'OK';
      const statusClass = isOk ? 'text-komplit' : 'text-danger';
      const statusLabel = isOk ? '✓ Selesai Sesuai' : 'Defect (Temuan)';
      const fillClass = isOk ? 'bar-fill green' : 'bar-fill';

      // Buat Kartu Baru
      const newCard = document.createElement('article');
      newCard.className = 'item-card';
      newCard.innerHTML = `
        <div class="item-card-top">
          <span class="badge-category badge-electrical">${badgeText}</span>
          <span class="status-text ${statusClass}">${statusLabel}</span>
        </div>
        <div class="item-card-name">${itemTitle}</div>
        <div class="item-card-meta">
          <span>SPV: ${spv}</span>
          <b>Progres: ${progres}%</b>
        </div>
        <div class="bar-track">
          <div class="${fillClass}" style="width: ${progres}%;"></div>
        </div>
      `;

      // Masukkan ke daftar paling atas
      const list = document.getElementById('checklistCardList');
      list.insertBefore(newCard, list.firstChild);

      closeChecklistModal();
      document.getElementById('formCatatan').value = '';
    }
  </script>
